const SPREADSHEET_ID = '1Q6ZF4OlsACKKq_4qksjg3-Ku25MxXXC-KBHy7ef7CeQ';
const SHEET_PENDAPATAN = 'pendapatan';
const SHEET_PENGELUARAN = 'pengeluaran';
const SHEET_KATEGORI = 'kategori';
const SHEET_SUMBER_DANA = 'sumber_dana'; // BARU

// ==========================================
// ROUTER doGet
// ==========================================
function doGet(e) {
  try {
    const action = (e && e.parameter && e.parameter.action) || '';
    switch (action) {
      case 'getAll': return getAllData();
      case 'getKategori': return getAllKategori();
      case 'getSumberDana': return getAllSumberDana(); // BARU
      case 'ping':
      case '': return jsonResponse({ ok: true, service: 'buku-keuangan', ready: true });
      default: return jsonResponse({ ok: false, error: 'Action tidak dikenal: ' + action });
    }
  } catch (error) {
    return jsonResponse({ ok: false, error: error.toString() });
  }
}

// ==========================================
// ROUTER doPost
// ==========================================
function doPost(e) {
  let lock = null;
  try {
    lock = LockService.getScriptLock();
    lock.waitLock(15000);
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse({ ok: false, error: 'Body request kosong' });
    }
    let data;
    try { data = JSON.parse(e.postData.contents); }
    catch (err) { return jsonResponse({ ok: false, error: 'Body bukan JSON valid' }); }

    switch (data.action) {
      case 'create': return createTransaction(data);
      case 'delete': return deleteTransaction(data.id, data.jenis);
      case 'createKategori': return createKategori(data);
      case 'deleteKategori': return deleteKategori(data.id);
      case 'createSumberDana': return createSumberDana(data);   // BARU
      case 'deleteSumberDana': return deleteSumberDana(data.id); // BARU
      default: return jsonResponse({ ok: false, error: 'Action tidak dikenal: ' + data.action });
    }
  } catch (error) {
    return jsonResponse({ ok: false, error: error.toString() });
  } finally {
    if (lock) { try { lock.releaseLock(); } catch (e) {} }
  }
}

// ==========================================
// UTIL
// ==========================================
function getSheet(name) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName(name);
  if (!sheet) throw new Error('Sheet "' + name + '" tidak ditemukan.');
  return sheet;
}

function formatTanggalCell(cell) {
  if (cell instanceof Date && !isNaN(cell.getTime())) {
    return `${cell.getFullYear()}-${String(cell.getMonth()+1).padStart(2,'0')}-${String(cell.getDate()).padStart(2,'0')}`;
  }
  return String(cell || '').trim();
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function generateId() {
  return Date.now() * 1000 + Math.floor(Math.random() * 1000);
}

// ==========================================
// GET: SEMUA TRANSAKSI (dengan sumber_dana)
// ==========================================
function getAllData() {
  const pendapatan = readTransactions(SHEET_PENDAPATAN, 'pendapatan');
  const pengeluaran = readTransactions(SHEET_PENGELUARAN, 'pengeluaran');
  const all = pendapatan.concat(pengeluaran);
  all.sort(function (a, b) {
    if (a.tanggal > b.tanggal) return -1;
    if (a.tanggal < b.tanggal) return 1;
    return b.id - a.id;
  });
  return jsonResponse({ ok: true, data: all });
}

function readTransactions(sheetName, jenis) {
  const sheet = getSheet(sheetName);
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  // 6 kolom: id, tanggal, uraian, nominal, kategori, sumber_dana
  const values = sheet.getRange(2, 1, lastRow - 1, 6).getValues();
  return values
    .filter(function (row) {
      return row[0] !== '' && row[0] !== null && row[0] !== undefined;
    })
    .map(function (row, index) {
      return {
        id: Number(row[0]) || (Date.now() + index),
        tanggal: formatTanggalCell(row[1]),
        uraian: String(row[2] || '').trim(),
        nominal: Number(row[3]) || 0,
        kategori: String(row[4] || 'Lainnya').trim(),
        sumber_dana: String(row[5] || '').trim(),
        jenis: jenis
      };
    });
}

// ==========================================
// GET: KATEGORI
// ==========================================
function getAllKategori() {
  const sheet = getSheet(SHEET_KATEGORI);
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return jsonResponse({ ok: true, data: [] });
  const values = sheet.getRange(2, 1, lastRow - 1, 3).getValues();
  const data = values
    .filter(function (row) {
      return row[0] !== '' && row[0] !== null && row[0] !== undefined && String(row[1]).trim() !== '';
    })
    .map(function (row) {
      return { id: Number(row[0]), nama: String(row[1]).trim(), jenis: String(row[2]).trim() };
    });
  return jsonResponse({ ok: true, data: data });
}

// ==========================================
// GET: SUMBER DANA
// ==========================================
function getAllSumberDana() {
  const sheet = getSheet(SHEET_SUMBER_DANA);
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return jsonResponse({ ok: true, data: [] });
  const values = sheet.getRange(2, 1, lastRow - 1, 3).getValues();
  const data = values
    .filter(function (row) {
      return row[0] !== '' && row[0] !== null && row[0] !== undefined && String(row[1]).trim() !== '';
    })
    .map(function (row) {
      return {
        id: Number(row[0]),
        nama: String(row[1]).trim(),
        saldo_awal: Number(row[2]) || 0
      };
    });
  return jsonResponse({ ok: true, data: data });
}

// ==========================================
// POST: TAMBAH TRANSAKSI
// ==========================================
function createTransaction(data) {
  if (data.jenis !== 'pendapatan' && data.jenis !== 'pengeluaran') {
    return jsonResponse({ ok: false, error: 'Jenis tidak valid' });
  }
  if (!data.tanggal) return jsonResponse({ ok: false, error: 'Tanggal wajib diisi' });
  if (!data.uraian || String(data.uraian).trim() === '') return jsonResponse({ ok: false, error: 'Uraian wajib diisi' });
  const nominal = Number(data.nominal);
  if (!nominal || nominal <= 0 || isNaN(nominal)) return jsonResponse({ ok: false, error: 'Nominal harus > 0' });
  if (!data.kategori || String(data.kategori).trim() === '') return jsonResponse({ ok: false, error: 'Kategori wajib diisi' });
  const tglStr = String(data.tanggal).trim();
  if (!/^\d{4}-\d{2}-\d{2}$/.test(tglStr)) return jsonResponse({ ok: false, error: 'Format tanggal harus YYYY-MM-DD' });

  const sheetName = data.jenis === 'pendapatan' ? SHEET_PENDAPATAN : SHEET_PENGELUARAN;
  const sheet = getSheet(sheetName);
  const newId = data.id && Number(data.id) > 0 ? Number(data.id) : generateId();

  sheet.appendRow([
    newId,
    tglStr,
    String(data.uraian).trim(),
    Math.round(nominal),
    String(data.kategori).trim(),
    String(data.sumber_dana || '').trim()
  ]);

  return jsonResponse({ ok: true, message: 'Transaksi ditambahkan', id: newId });
}

// ==========================================
// POST: HAPUS TRANSAKSI
// ==========================================
function deleteTransaction(id, jenis) {
  if (jenis !== 'pendapatan' && jenis !== 'pengeluaran') {
    return jsonResponse({ ok: false, error: 'Jenis tidak valid' });
  }
  const numericId = Number(id);
  if (!numericId) return jsonResponse({ ok: false, error: 'ID tidak valid' });
  const sheetName = jenis === 'pendapatan' ? SHEET_PENDAPATAN : SHEET_PENGELUARAN;
  const sheet = getSheet(sheetName);
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return jsonResponse({ ok: false, error: 'Tidak ada data' });
  const ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
  for (let i = 0; i < ids.length; i++) {
    if (Number(ids[i][0]) === numericId) {
      sheet.deleteRow(i + 2);
      return jsonResponse({ ok: true, message: 'Transaksi dihapus' });
    }
  }
  return jsonResponse({ ok: false, error: 'ID tidak ditemukan' });
}

// ==========================================
// POST: KATEGORI
// ==========================================
function createKategori(data) {
  const nama = String(data.nama || '').trim();
  const jenis = String(data.jenis || '').trim();
  if (!nama) return jsonResponse({ ok: false, error: 'Nama kategori wajib diisi' });
  if (nama.length > 30) return jsonResponse({ ok: false, error: 'Nama maksimal 30 karakter' });
  if (jenis !== 'pendapatan' && jenis !== 'pengeluaran') return jsonResponse({ ok: false, error: 'Jenis tidak valid' });

  const sheet = getSheet(SHEET_KATEGORI);
  const lastRow = sheet.getLastRow();
  let newId = 1;
  if (lastRow >= 2) {
    const existing = sheet.getRange(2, 1, lastRow - 1, 3).getValues();
    const ids = existing.map(function (r) { return Number(r[0]) || 0; });
    newId = Math.max.apply(null, ids.concat([0])) + 1;
    for (let i = 0; i < existing.length; i++) {
      if (String(existing[i][1]).trim().toLowerCase() === nama.toLowerCase() &&
          String(existing[i][2]).trim() === jenis) {
        return jsonResponse({ ok: false, error: 'Kategori "' + nama + '" sudah ada untuk jenis ' + jenis });
      }
    }
  }
  sheet.appendRow([newId, nama, jenis]);
  return jsonResponse({ ok: true, message: 'Kategori ditambahkan', id: newId });
}

function deleteKategori(id) {
  const numericId = Number(id);
  if (!numericId) return jsonResponse({ ok: false, error: 'ID tidak valid' });
  const sheet = getSheet(SHEET_KATEGORI);

  const ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
  for (let i = 0; i < ids.length; i++) {
    if (Number(ids[i][0]) === numericId) {
      sheet.deleteRow(i + 2);
      return jsonResponse({ ok: true, message: 'Kategori dihapus' });
    }
  }
  return jsonResponse({ ok: false, error: 'ID tidak ditemukan' });
}

// SETUP (jalankan sekali dari editor)
// ==========================================
function setupSpreadsheet() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);

  // pendapatan
  let pendapatan = ss.getSheetByName(SHEET_PENDAPATAN);
  if (!pendapatan) pendapatan = ss.insertSheet(SHEET_PENDAPATAN);
  if (pendapatan.getLastRow() === 0) {
    pendapatan.appendRow(['id', 'tanggal', 'uraian', 'nominal', 'kategori', 'sumber_dana']);
    pendapatan.getRange('A1:F1').setFontWeight('bold').setBackground('#e8f0fe');
    pendapatan.setFrozenRows(1);
    pendapatan.getRange('B2:B').setNumberFormat('yyyy-mm-dd');
  }

  // pengeluaran
  let pengeluaran = ss.getSheetByName(SHEET_PENGELUARAN);
  if (!pengeluaran) pengeluaran = ss.insertSheet(SHEET_PENGELUARAN);
  if (pengeluaran.getLastRow() === 0) {
    pengeluaran.appendRow(['id', 'tanggal', 'uraian', 'nominal', 'kategori', 'sumber_dana']);
    pengeluaran.getRange('A1:F1').setFontWeight('bold').setBackground('#fce8e6');
    pengeluaran.setFrozenRows(1);
    pengeluaran.getRange('B2:B').setNumberFormat('yyyy-mm-dd');
  }

  // kategori
  let kategori = ss.getSheetByName(SHEET_KATEGORI);
  if (!kategori) kategori = ss.insertSheet(SHEET_KATEGORI);
  if (kategori.getLastRow() === 0) {
    kategori.appendRow(['id', 'nama', 'jenis']);
    kategori.getRange('A1:C1').setFontWeight('bold').setBackground('#fef7e0');
    kategori.setFrozenRows(1);
    [
      [1, 'Gaji', 'pendapatan'],
      [2, 'Bonus', 'pendapatan'],
      [3, 'Investasi', 'pendapatan'],
      [4, 'Lainnya', 'pendapatan'],
      [5, 'Makanan', 'pengeluaran'],
      [6, 'Transportasi', 'pengeluaran'],
      [7, 'Belanja', 'pengeluaran'],
      [8, 'Tagihan', 'pengeluaran'],
      [9, 'Pendidikan', 'pengeluaran'],
      [10, 'Kesehatan', 'pengeluaran'],
      [11, 'Lainnya', 'pengeluaran']
    ].forEach(function (r) { kategori.appendRow(r); });
  }

  // sumber_dana
  let sumber = ss.getSheetByName(SHEET_SUMBER_DANA);
  if (!sumber) sumber = ss.insertSheet(SHEET_SUMBER_DANA);
  if (sumber.getLastRow() === 0) {
    sumber.appendRow(['id', 'nama', 'saldo_awal']);
    sumber.getRange('A1:C1').setFontWeight('bold').setBackground('#e0f2fe');
    sumber.setFrozenRows(1);
    [
      [1, 'Tunai', 0],
      [2, 'Bank', 0],
      [3, 'E-Wallet', 0]
    ].forEach(function (r) { sumber.appendRow(r); });
  }

  SpreadsheetApp.getUi().alert('✅ Setup selesai!\n\nSheet: pendapatan, pengeluaran, kategori, sumber_dana sudah siap.');
}
